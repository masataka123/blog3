\begin{tcolorbox}[mybox]
\begin{defn}
$U\subset\mathbb{R}^d$を開集合, 
$\mathcal{B}(U)$を全ての開集合を含む最小の Borel $\sigma$代数とする. 

Radon measure' $\mu\colon\mathcal{B}(U)\to[0,\infty]$はBorel measureであり, 
\begin{enumerate}
\item (正則性) 任意の$ E\in\mathcal{B}(U)$について, 
\begin{equation*}
\mu(E)
=
\inf\{\mu(V)\mid E\subset V\subset U;\ \text{open}\}
=
\sup\{\mu(A)\mid A\subset E:\ \text{cpt}\}.
\end{equation*}
\item 任意のコンパクト集合$K\in\mathcal{B}(U)$について$\mu(K)<\infty$.
\end{enumerate}

complex Radon measure $\mu\colon\mathcal{B}(U)\to\mathbb{C}$を複素Borel測度でtotal variation
\begin{equation*}
\mathcal{B}(U)\longrightarrow[0,\infty] \quad E \longmapsto
|\mu|(E)
:=
\sup\left\{
\sum_{i=1}^{\infty}|\mu(E_i)| \mid E=\bigcup_{i=1}^{\infty}E_i, E_i\cap E_j=\varnothing (\forall i,j)
\right\}.
\end{equation*}
%$|\mu|\colon\mathcal{B}(U)\to[0,\infty]$
がRadon測度となることとする.
\end{defn}
\end{tcolorbox}
上はRudinの意味でのRadon measureと同じである. 
完全加法性が絶対収束になることを要請しており, この場合total variationが有限になること従う.

\noindent\underline{\textbf{Lem 11 (On trace)}}

\noindent$\bullet$\;
\begin{equation*}
T=i^{p^2}\sum_{J,K\in\{m;p\}}T_{J\bar K}\,dz^J\wedge d\bar z^K
\in\mathcal{D}'^{(p,p)}(U)
\end{equation*}
is a pos.\ current, where $U\subset\mathbb{C}^m$ is open.

Then
\begin{equation*}
\operatorname{Tr}(T):=\sum_{J\in\{m;p\}}T_{J\bar J}
\end{equation*}
is a pos.\ $(0,0)$-current.

\noindent\underline{prf.}\ It follows from [松澤2026-06-30, Prop 9].\hfill //

\noindent\underline{\textbf{Def 12 (Smoothing)}}

\noindent$\bullet$\; $h\colon\mathbb{C}^m\longrightarrow\mathbb{R}$: $C^\infty$-func.\ s.t.\
$h(z)\geq 0$ $(\forall z)$, $\operatorname{Supp}(h)\subset B(0;1)$,
\begin{equation*}
h(z)\text{ only dep.\ on }|z|,
\qquad
\int_{\mathbb{R}^m}h(z)\,dz=1.
\end{equation*}

\noindent$\bullet$\; $h_\epsilon\colon\mathbb{C}^m\longrightarrow\mathbb{R}$:
\begin{equation*}
h_\epsilon(z):=\frac{1}{\epsilon^m}h\left(\frac{z}{\epsilon}\right).
\end{equation*}

\noindent$\bullet$\;
\begin{equation*}
T=\sum_{\substack{J\in\{m;p\}\\K\in\{m;q\}}}
T_{J\bar K}\,dz^J\wedge d\bar z^K
\in\mathcal{D}'^{(p,q)}(U);
\quad\text{current}
\qquad
(U\subset\mathbb{C}^m;\text{ open}).
\end{equation*}

\begin{equation*}
\textcolor{blue}{
T_{J\bar K}\wedge
\left(\bigwedge_{\ell=1}^m dz^\ell\wedge d\bar z^\ell\right)
=
T\wedge dz^{J^\circ}\wedge d\bar z^{K^\circ}
\text{ で定義},
\qquad
(J^\circ,K^\circ:\text{ ord.\ compl.})
}
\end{equation*}

\noindent$\bullet$\;
\begin{equation*}
U_\epsilon:=
\left\{
z\in\mathbb{C}^m
\ \middle|\
\overline{B(z;\epsilon)}\subset U
\right\}.
\end{equation*}

Then
\begin{equation*}
T_\epsilon:=
\sum_{\substack{J\in\{m;p\}\\K\in\{m;q\}}}
(T_{J\bar K}*h_\epsilon)\,dz^J\wedge d\bar z^K
\end{equation*}
is an elem.\ of $\mathcal{D}'^{(p,q)}(U_\epsilon)$, where
\begin{equation*}
T_{J\bar K}*h_\epsilon
=
\left[
z\longmapsto T_{J\bar K}\bigl(h_\epsilon(u-z)\bigr)
\right],
\qquad\text{i.e.,}
\end{equation*}
\begin{equation*}
(T_{J\bar K}*h_\epsilon)(f)
:=
\int_{U_\epsilon}
\left(T_{J\bar K}\bigl(h_\epsilon(u-z)\bigr)\right)_z
\cdot f(z)\,dz,
\end{equation*}
\begin{equation*}
\textcolor{blue}{
u\text{：変数},
\qquad
z\text{：変数},
\qquad
dz=dz_1\wedge d\bar z_1\wedge\cdots\wedge dz_m\wedge d\bar z_m
}
\end{equation*}
\begin{align*}
&=
T_{J\bar K}
\left(
\int_{\mathbb{C}^m}f(z)h_\epsilon(u-z)\,dz
\right)
=
T_{J\bar K}(f*h_\epsilon).
\end{align*}
\textcolor{blue}{$u$を変数とする関数}

We call $T_\epsilon$ a ``smoothing'' of $T$
\[
\textcolor{red}{[判読不能：この箇所は使用禁止の数式区切りを避けるため次行に転記]}
\]
\begin{equation*}
\left(
z\longmapsto T_{J\bar K}\bigl(h_\epsilon(u-z)\bigr);
\ C^\infty\text{-func.}
\right).
\end{equation*}

% [page 2]
\noindent\underline{\textbf{Rem 13}}

\noindent$\bullet$\; $H_{J\bar K,\epsilon}$: func.\ corr.\ to
\begin{equation*}
z\longmapsto T_{J\bar K}\bigl(h_\epsilon(u-z)\bigr).
